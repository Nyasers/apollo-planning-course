# 轨迹指标提取脚本

> 使用 Python 解析 Apollo cyber record，提取 Planning 轨迹中的速度、加速度、Jerk 等指标，用于体感评估。

---

## 一、环境准备

在**容器内**（`aem enter` 后）执行，Apollo 容器已内置 `cyber_py`：

```bash
# 验证 Python cyber 接口可用
python3 -c "from cyber.python.cyber_py3 import cyber; print('OK')"
```

## 二、完整提取脚本

将以下脚本保存为 `/apollo_workspace/extract_metrics.py`：

```python
#!/usr/bin/env python3
"""
Apollo Planning 轨迹体感指标提取脚本
用法: python3 extract_metrics.py <record_file>
"""

import sys
import math
import numpy as np
from cyber.python.cyber_py3 import cyber
from cyber.python.cyber_py3.record import RecordReader
from modules.planning.proto.planning_pb2 import ADCTrajectory


def extract_trajectory_metrics(record_file: str):
    """从 cyber record 提取轨迹速度、加速度、Jerk 指标"""
    reader = RecordReader(record_file)

    speeds = []        # 速度 m/s
    accels = []        # 规划加速度 m/s²
    timestamps = []    # 时间戳 s

    print(f"正在读取: {record_file}")

    for msg in reader.read_messages("/apollo/planning/trajectory"):
        traj = ADCTrajectory()
        traj.ParseFromString(msg.message)

        # 取每帧轨迹的第一个点（当前规划起点）
        if traj.trajectory_point:
            pt = traj.trajectory_point[0]
            speeds.append(pt.v)
            accels.append(pt.a)
            timestamps.append(msg.timestamp / 1e9)  # ns → s

    if not speeds:
        print("未找到 /apollo/planning/trajectory 数据，请确认 record 文件正确")
        return

    speeds = np.array(speeds)
    accels = np.array(accels)
    timestamps = np.array(timestamps)

    # 计算 Jerk（加速度的时间导数）
    dt = np.diff(timestamps)
    dt = np.where(dt < 1e-6, 1e-6, dt)  # 防止除零
    jerks = np.diff(accels) / dt

    # 计算横向加速度（需要曲率数据，此处用加速度近似）
    # 完整版需读取 localization/pose 中的 angular_velocity
    lat_accels = np.zeros_like(speeds)  # 占位，完整实现见下方说明

    print("\n===== 体感指标统计 =====")
    print(f"数据帧数:        {len(speeds)}")
    print(f"仿真时长:        {timestamps[-1] - timestamps[0]:.1f} s")
    print(f"平均速度:        {np.mean(speeds):.2f} m/s  ({np.mean(speeds)*3.6:.1f} km/h)")
    print(f"最大速度:        {np.max(speeds):.2f} m/s")
    print()
    print(f"纵向加速度 max:  {np.max(np.abs(accels)):.3f} m/s²  (阈值: 3.0)")
    print(f"纵向加速度 RMS:  {np.sqrt(np.mean(accels**2)):.3f} m/s²")
    print(f"超限帧数 (>3.0): {np.sum(np.abs(accels) > 3.0)}")
    print()
    print(f"纵向 Jerk max:   {np.max(np.abs(jerks)):.3f} m/s³  (阈值: 4.0)")
    print(f"纵向 Jerk RMS:   {np.sqrt(np.mean(jerks**2)):.3f} m/s³")
    print(f"超限帧数 (>4.0): {np.sum(np.abs(jerks) > 4.0)}")
    print()
    hard_brake = np.sum(accels < -4.0)
    print(f"急刹车次数:      {hard_brake}  (减速度 > 4.0 m/s²)")

    # 输出超限时刻（方便在 DreamView 中定位）
    if np.any(np.abs(accels) > 3.0):
        idxs = np.where(np.abs(accels) > 3.0)[0]
        print("\n纵向加速度超限时刻 (前5个):")
        for i in idxs[:5]:
            print(f"  t={timestamps[i]:.2f}s  a={accels[i]:.3f} m/s²  v={speeds[i]:.2f} m/s")

    return {
        "speeds": speeds,
        "accels": accels,
        "jerks": jerks,
        "timestamps": timestamps,
    }


if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("用法: python3 extract_metrics.py <record_file>")
        print("示例: python3 extract_metrics.py data/bag/my_record.record")
        sys.exit(1)

    extract_trajectory_metrics(sys.argv[1])
```

## 四、不使用 cyber record 的简化版

如果没有录包，可以直接分析 Planning 文本日志中的速度字段：

```python
#!/usr/bin/env python3
"""简化版：从 Planning 日志提取速度"""
import re
import numpy as np

log_file = "/apollo_workspace/data/log/planning.INFO"
speeds = []

with open(log_file) as f:
    for line in f:
        # 匹配日志中的速度字段（格式因版本而异）
        m = re.search(r'speed[=:]\s*([\d.]+)', line)
        if m:
            speeds.append(float(m.group(1)))

if speeds:
    arr = np.array(speeds)
    print(f"平均速度: {arr.mean():.2f} m/s")
    print(f"最大速度: {arr.max():.2f} m/s")
```
