## 二、完整可视化脚本

将以下内容保存为 `/apollo_workspace/plot_metrics.py`，配合 `extract_metrics.py` 使用：

```python
#!/usr/bin/env python3
"""
Apollo 体感指标可视化脚本
依赖 extract_metrics.py 中的 extract_trajectory_metrics 函数
用法: python3 plot_metrics.py <record_file>
"""

import sys
import numpy as np
import matplotlib
matplotlib.use('Agg')  # 无显示器时使用（容器内）
import matplotlib.pyplot as plt
import matplotlib.gridspec as gridspec

# 引用提取脚本（两个文件放同一目录）
sys.path.insert(0, '/apollo_workspace')
from extract_metrics import extract_trajectory_metrics


def plot_metrics(record_file: str, output_path: str = "metrics_report.png"):
    data = extract_trajectory_metrics(record_file)
    if data is None:
        return

    t = data["timestamps"] - data["timestamps"][0]  # 归零
    v = data["speeds"]
    a = data["accels"]
    j = data["jerks"]
    t_j = t[1:]  # jerk 比其他少一个点

    # 阈值线
    LON_ACCEL_MAX = 3.0
    JERK_MAX = 4.0
    HARD_BRAKE = -4.0

    fig = plt.figure(figsize=(14, 10))
    fig.suptitle(f"Apollo Planning 体感指标报告\n{record_file}", fontsize=12)
    gs = gridspec.GridSpec(3, 1, hspace=0.4)

    # 子图1：速度
    ax1 = fig.add_subplot(gs[0])
    ax1.plot(t, v, 'b-', linewidth=1.5, label='速度 (m/s)')
    ax1.set_ylabel('速度 (m/s)')
    ax1.set_title('速度曲线')
    ax1.legend(loc='upper right')
    ax1.grid(True, alpha=0.3)

    # 子图2：纵向加速度
    ax2 = fig.add_subplot(gs[1])
    ax2.plot(t, a, 'g-', linewidth=1.5, label='纵向加速度')
    ax2.axhline(y=LON_ACCEL_MAX, color='r', linestyle='--', alpha=0.7, label=f'上限 ±{LON_ACCEL_MAX}')
    ax2.axhline(y=-LON_ACCEL_MAX, color='r', linestyle='--', alpha=0.7)
    ax2.axhline(y=HARD_BRAKE, color='darkred', linestyle=':', alpha=0.7, label=f'急刹车 {HARD_BRAKE}')
    # 高亮超限区域
    exceed_mask = np.abs(a) > LON_ACCEL_MAX
    ax2.fill_between(t, a, 0, where=exceed_mask, color='red', alpha=0.3, label='超限')
    ax2.set_ylabel('加速度 (m/s²)')
    ax2.set_title(f'纵向加速度（超限帧: {exceed_mask.sum()}）')
    ax2.legend(loc='upper right', fontsize=8)
    ax2.grid(True, alpha=0.3)

    # 子图3：Jerk
    ax3 = fig.add_subplot(gs[2])
    ax3.plot(t_j, j, 'm-', linewidth=1.5, label='纵向 Jerk')
    ax3.axhline(y=JERK_MAX, color='r', linestyle='--', alpha=0.7, label=f'上限 ±{JERK_MAX}')
    ax3.axhline(y=-JERK_MAX, color='r', linestyle='--', alpha=0.7)
    exceed_j = np.abs(j) > JERK_MAX
    ax3.fill_between(t_j, j, 0, where=exceed_j, color='red', alpha=0.3, label='超限')
    ax3.set_xlabel('时间 (s)')
    ax3.set_ylabel('Jerk (m/s³)')
    ax3.set_title(f'纵向 Jerk（超限帧: {exceed_j.sum()}）')
    ax3.legend(loc='upper right', fontsize=8)
    ax3.grid(True, alpha=0.3)

    plt.savefig(output_path, dpi=120, bbox_inches='tight')
    print(f"\n图表已保存到: {output_path}")
    print("提示：可用 scp 或共享目录将图片传到宿主机查看")


if __name__ == "__main__":
    if len(sys.argv) < 2:
        print("用法: python3 plot_metrics.py <record_file> [output.png]")
        sys.exit(1)

    record = sys.argv[1]
    output = sys.argv[2] if len(sys.argv) > 2 else "metrics_report.png"
    plot_metrics(record, output)
```

## 四、DreamView 中的轨迹可视化（无需代码）

在 DreamView 中可以直接观察规划轨迹：

1. 打开 DreamView（`http://localhost:8888`）
2. 点击左侧 **Layer Menu** → 开启 **Planning** 层
3. 轨迹会以彩色曲线显示，颜色通常代表速度高低
4. 回放 cyber record 时轨迹随时间变化，可暂停观察关键帧

**DreamView 中可观察的信号**：
- **轨迹点颜色**：绿色=正常速度，黄色=较慢，红色=停止或急刹
- **速度图表**：DreamView+ 右侧面板显示实时速度/加速度曲线
- **模块状态**：顶部面板显示 Planning 是否正常运行

## 五、图表解读方法

| 现象 | 含义 | 优化方向 |
|------|------|---------|
| 加速度曲线有尖峰（红色高亮） | 急加速或急刹车 | 调整加速度上限、提前减速 |
| Jerk 曲线剧烈震荡 | 速度控制不平滑 | 增大速度平滑窗口、调整 jerk_bound |
| 速度在某段持续为 0 | 车辆停滞 | 检查该时刻的 Planning 日志和障碍物状态 |
| 速度曲线有锯齿状 | 频繁刹车-加速循环 | 优化跟车距离参数、减少 stop/go 行为 |
